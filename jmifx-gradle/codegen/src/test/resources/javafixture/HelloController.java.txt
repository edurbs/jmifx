package fixture;

import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;

/**
 * Test controller — compiled only in Task 6's integration test. Fields are
 * package-visible (jmifx convention: generated code assigns them directly, no
 * reflection). The @FXML annotation is optional for jmifx and omitted here so
 * this file has no javafx.fxml dependency.
 */
public class HelloController {

    Label greetingLabel;
    TextField nameField;
    Button greetButton;

    public final java.util.List<String> events = new java.util.ArrayList<>();

    public void initialize() {
        if (greetingLabel != null) {
            events.add("init:greetingLabel");
        }
        if (nameField != null) {
            events.add("init:nameField");
        }
        if (greetButton != null) {
            events.add("init:greetButton");
        }
    }

    public void greet(javafx.event.ActionEvent event) {
        events.add("greet");
    }
}
